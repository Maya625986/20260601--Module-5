import unittest
import pandas as pd
from pandas.testing import assert_frame_equal

class TestDataFrames(unittest.TestCase):

    def test_dataframes_equal(self):
        df1 = pd.DataFrame({
            "A": [1, 2],
            "B": [3, 4]
        })

        df2 = pd.DataFrame({
            "A": [1, 2],
            "B": [3, 4]
        })

        assert_frame_equal(df1, df2)

        if __name__ == "__main__":
            unittest.main()
